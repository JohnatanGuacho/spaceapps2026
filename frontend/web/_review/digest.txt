export function AssistantDock() {
  const env = useActiveEnv();
  const { t } = useLang();
  const booted = useStore(appStore, (s) => s.booted);
  const chatOpen = useStore(appStore, (s) => s.chatOpen);
  const welcomed = useStore(appStore, (s) => s.welcomed);
  const floating = env.section === "home" || env.section === "module";
  const key = `${env.section}:${env.module?.id ?? ""}`;
  // Sección cuyo mensaje ya se recogió y sección cuyo mensaje se pidió con un clic.
  const [hiddenKey, setHiddenKey] = useState<string | null>(null);
  const [askedKey, setAskedKey] = useState<string | null>(null);
  const [introDone, setIntroDone] = useState(false);
  // En las vistas 3D el mensaje aparece solo; en las pantallas de trabajo
  // espera a que pulsen la mascota, para no tapar controles ni gráficas.
  const auto = floating || env.section === "cabin";
  const bubble = askedKey === key || (auto && hiddenKey !== key);
  const intro = booted && !introDone;
  
  // La entrada (la llama cae y se vuelve esfera) se reproduce al terminar la carga.
  useEffect(() => {
  if (!booted) return;
  const id = setTimeout(() => setIntroDone(true), 2400);
  return () => clearTimeout(id);
  }, [booted]);
  
  // Cada sección trae un mensaje nuevo, que se recoge solo tras unos segundos.
  useEffect(() => {
  if (!booted || (env.section === "home" && !welcomed)) return;
  const id = setTimeout(() => {
  setHiddenKey(key);
  setAskedKey(null);
  }, env.section === "home" ? 12000 : 9000);
  return () => clearTimeout(id);
  }, [key, env.section, welcomed, booted, askedKey]);
  
  if (!booted) return null;
  const message = env.section === "module" ? env.module!.guide : env.section === "cabin" ? env.module!.cabinGuide : TIPS[env.section];
  const dismiss = (): void => {
  setHiddenKey(key);
  setAskedKey(null);
  if (env.section === "home") setPrefs({ welcomed: true });
  };
  const open = (): void => appStore.set({ chatOpen: true });
  
  return (
  <>
  <div className={`${styles.dock} ${floating ? styles.floating : styles.docked} ${env.section === "cabin" ? styles.cabin : ""}`}>
  <button type="button" className={styles.mascot} onClick={() => (bubble ? open() : setAskedKey(key))} aria-label={t(l("Asistente FUEGO", "FUEGO assistant"))}>

export function frontier(f: FrontierModel, a: FrontierInput): FrontierResult {
  const U = Math.max(0, a.U);
  const quench = f.quenchA * Math.exp(-U / f.quenchU);
  const rise = f.gravityRise * Math.pow(Math.max(0, (U - f.uStar) / (U_BUOY_1G - f.uStar)), 1.5);
  const blow = 8 * Math.pow(Math.max(0, U - 120) / 120, 2);
  const pf = Math.pow(101.3 / clamp(a.P, 15, 320), f.pExp);
  const agents = 1 + 1.62 * ((a.co2 ?? 0) / 100) + 0.4 * ((a.he ?? 0) / 100) + 2.5 * (a.mist ?? 0);
  const heat = f.heatDrop * sat((a.heat ?? 0) / 250);
  const loc = (f.locMin + quench + rise + blow) * pf * agents - heat;
  const partial = a.g > 0.02 && a.g < 0.9;
  const lo = loc - f.sigma - (partial ? f.lunarComp : 0);
  const hi = loc + f.sigma;
  return { loc, lo, hi };
  }

export function spreadRate(m: Material, s: SpreadInput): number {
  if (m.form === "droplet" || m.form === "wire") return 0;
  const f = Math.pow(s.o2 / 21.7, 1.6) * Math.pow(s.P / 101.3, 0.3) * sat(s.M / 1.5) * m.spreadK;
  const Uc = Math.max(s.U, 0.5);
  const base = s.mode === "opposed" ? 0.13 * (Uc / 20) * Math.exp(1 - Uc / 20) : 0.009 * Uc;
  const buoy = s.mode === "opposed" ? 1 : 1 + 2 * Math.cbrt(Math.max(s.g, 0));
  return f * base * buoy;
  }

export function dropletK(d: DropletModel, o2: number, P: number, U: number, co2 = 0, he = 0): number {
  const b = d.b21 * (o2 / 21);
  const kO2 = Math.log(1 + b) / Math.log(1 + d.b21);
  // La convección solo acelera el quemado por encima del flujo residual de microgravedad.
  const conv = 1 + 0.3 * Math.sqrt(Math.max(0, U - 0.5) / 10);
  const dil = (1 - 0.3 * (co2 / 100)) * (1 + 0.6 * (he / 100));
  return d.k21 * kO2 * conv * dil * Math.pow(P / 101.3, 0.1);
  }

export type Criterion = "atmosphere" | "gravity" | "scale" | "data" | "pillar";
export type RankWeights = Record<Criterion, number>;
export const CRITERIA: Criterion[] = ["atmosphere", "gravity", "scale", "data", "pillar"];
export const DEFAULT_WEIGHTS: RankWeights = { atmosphere: 3, gravity: 3, scale: 1, data: 2, pillar: 2 };
  
export const CRITERION_LABEL: Record<Criterion, L> = {
  atmosphere: l("Atmósfera (O₂ y presión)", "Atmosphere (O₂ and pressure)"),
  gravity: l("Régimen de gravedad", "Gravity regime"),
  scale: l("Escala del ensayo", "Test scale"),
  data: l("Disponibilidad de datos", "Data availability"),
  pillar: l("Pilar de seguridad", "Safety pillar"),
  };
export interface RankQuery {
  o2: number;
  p: number;
  g: number;
  pillar?: Pillar | null;
  }
export interface Ranked {
  inv: Investigation;
  /** Puntaje 0–1 (1 = más relevante). */
  score: number;
  /** Cercanía por criterio (1 = coincide). */
  parts: RankWeights;
  /** Fracción de atributos numéricos publicados (O₂, presión, flujo). */
  coverage: number;
  }
export function rankInvestigations(q: RankQuery, w: RankWeights = DEFAULT_WEIGHTS, list: Investigation[] = INVESTIGATIONS): Ranked[] {
  const total = CRITERIA.reduce((a, c) => a + w[c], 0) || 1;
  return list
  .map((inv) => {
  const d = distances(inv, q);
  const dist = CRITERIA.reduce((a, c) => a + w[c] * d[c], 0) / total;
  const parts = Object.fromEntries(CRITERIA.map((c) => [c, 1 - d[c]])) as RankWeights;
  const coverage = [inv.o2, inv.p, inv.flow].filter(Boolean).length / 3;
  return { inv, score: 1 - dist, parts, coverage };
  })
  .sort((a, b) => b.score - a.score);
  }

export function useScenario(m: ModuleDef): {
  scenario: SceneScenario;
  set: (patch: ScenarioPatch) => void;
  go: (patch: ScenarioPatch) => void;
  href: (patch: ScenarioPatch) => string;
  } {
  const sp = useSearchParams();
  const pathname = usePathname();
  const router = useRouter();
  const query = sp.toString();
  const scenario = useMemo(() => parseScenario(m, new URLSearchParams(query)), [m, query]);
  const href = useCallback((patch: ScenarioPatch) => pathname + patchQuery(query, patch), [pathname, query]);
  const set = useCallback(
  (patch: ScenarioPatch) => {
  // La URL real puede ir por delante del render: se parte de window.location.
  window.history.replaceState(null, "", window.location.pathname + patchQuery(window.location.search.slice(1), patch));
  },
  [],
  );
  const go = useCallback((patch: ScenarioPatch) => router.push(href(patch)), [router, href]);
  return { scenario, set, go, href };
  }

export function patchQuery(current: string, patch: ScenarioPatch): string {
  const sp = new URLSearchParams(current);

  if (v === null || v === undefined || v === "") sp.delete(k);

export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
  a = (a + 0x6d2b79f5) >>> 0;
  let t = a;
  t = Math.imul(t ^ (t >>> 15), t | 1);
  t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
